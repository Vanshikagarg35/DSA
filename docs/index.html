019. Median of two sorted arrays

What it asks: Find the median of two sorted arrays in logarithmic time.

Source: Standard C++17 solution (not present in the repository).

Traditional approach: binary-search partition of the smaller array.

class Solution {
public:
    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
        // Binary-search the shorter array.
        if (nums1.size() > nums2.size()) return findMedianSortedArrays(nums2, nums1);

        int m = nums1.size();
        int n = nums2.size();
        int left = 0, right = m;

        while (left <= right) {
            int cut1 = left + (right - left) / 2;
            int cut2 = (m + n + 1) / 2 - cut1;

            int left1 = (cut1 == 0) ? INT_MIN : nums1[cut1 - 1];
            int right1 = (cut1 == m) ? INT_MAX : nums1[cut1];
            int left2 = (cut2 == 0) ? INT_MIN : nums2[cut2 - 1];
            int right2 = (cut2 == n) ? INT_MAX : nums2[cut2];

            if (left1 <= right2 && left2 <= right1) {
                if ((m + n) % 2 == 1) return max(left1, left2);
                return (max(left1, left2) + min(right1, right2)) / 2.0;
            }
            if (left1 > right2) right = cut1 - 1;
            else left = cut1 + 1;
        }
        return 0.0; // inputs are valid, so this line is never reached
    }
};

long long maxSumOfSizeK(const vector<int>& nums, int k) {
    if (k <= 0 || k > (int)nums.size()) return 0;

    long long windowSum = 0;
    for (int i = 0; i < k; ++i) windowSum += nums[i];
    long long maximumSum = windowSum;

    for (int right = k; right < (int)nums.size(); ++right) {
        windowSum += nums[right];
        windowSum -= nums[right - k];
        maximumSum = max(maximumSum, windowSum);
    }

031. Minimum window substring

What it asks: Find the smallest substring containing all characters of a target.

Source: Exact repository code - 0076-minimum-window-substring/0076-minimum-window-substring.cpp

class Solution {
public:
    string minWindow(string s, string t) {
        if (s.length() < t.length()) return "";

        unordered_map<char, int> mp;
        for (char c : t) {
            mp[c]++;
        }

        int required = mp.size();
        int i = 0, j = 0;
        int minLen = INT_MAX;
        int startIdx = 0;

        while (j < s.length()) {
            if (mp.find(s[j]) != mp.end()) {
                mp[s[j]]--;
                if (mp[s[j]] == 0) {
                    required--;
                }
            }

            while (required == 0) {
                if (j - i + 1 < minLen) {
                    minLen = j - i + 1;
                    startIdx = i;
                }

                if (mp.find(s[i]) != mp.end()) {
                    mp[s[i]]++;
                    if (mp[s[i]] > 0) {
                        required++;
                    }
                }
                i++;
            }
            j++;
        }

        if (minLen == INT_MAX) return "";
        return s.substr(startIdx, minLen);
    }
};

What it asks: Design a fixed-capacity cache that evicts the least frequently used key.

    int capacity, minFrequency = 0;
    unordered_map<int, int> value, frequency;
    unordered_map<int, list<int>> keysAtFrequency;
    unordered_map<int, list<int>::iterator> position;

        keysAtFrequency[oldFrequency].erase(position[key]);
        if (keysAtFrequency[oldFrequency].empty() && minFrequency == oldFrequency) {
            ++minFrequency;
        }
        ++frequency[key];
        keysAtFrequency[frequency[key]].push_front(key);
        position[key] = keysAtFrequency[frequency[key]].begin();
    }

    int get(int key) {
        if (!value.count(key)) return -1;
        increaseFrequency(key);
        return value[key];
    }
    void put(int key, int newValue) {
        if (capacity == 0) return;
        if (value.count(key)) {
            value[key] = newValue;
            increaseFrequency(key);
            return;
        }
        if ((int)value.size() == capacity) {
            int keyToRemove = keysAtFrequency[minFrequency].back();
            keysAtFrequency[minFrequency].pop_back();
            value.erase(keyToRemove);
            frequency.erase(keyToRemove);
            position.erase(keyToRemove);

        value[key] = newValue;
        frequency[key] = 1;
        minFrequency = 1;
        keysAtFrequency[1].push_front(key);
        position[key] = keysAtFrequency[1].begin();
    }

152. Diameter of binary tree

What it asks: Find the longest path between any two nodes in a binary tree.

class Solution {
public:
    int countDia(TreeNode* root, int &maxDia) {
        if(root == NULL){
            return 0;
        }
        
        int leftCount = countDia(root->left, maxDia);
        int rightCount = countDia(root->right, maxDia);
        
        maxDia = max(maxDia, leftCount + rightCount);
        
        return max(leftCount, rightCount) + 1;
    }

    int diameterOfBinaryTree(TreeNode* root) {
        int maxDia = 0;
        countDia(root, maxDia);
        return maxDia;
    }
};

153. Right side view

What it asks: Return the tree node visible from the right at each depth.

Source: Exact repository code - 0199-binary-tree-right-side-view/0199-binary-tree-right-side-view.cpp

class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
        if(!root)
            return {};
        
        queue<TreeNode*> que;
        vector<int> result;
        
        que.push(root);
        
        while(!que.empty()) {
            int n = que.size();
            TreeNode* rightNode = NULL;
            while(n--) {
                rightNode = que.front();
                que.pop();
                
                if(rightNode->left)
                    que.push(rightNode->left);
                if(rightNode->right)
                    que.push(rightNode->right);
            }
            result.push_back(rightNode->val);
        }
        return result;
    }
};

154. Bottom view

What it asks: Return the lowest visible tree node at every horizontal distance.

Source: Standard C++17 solution (not present in the repository).

Traditional approach: BFS; overwrite the value for every horizontal distance.

vector<int> bottomView(Node* root) {
    if (!root) return {};
    map<int, int> valueAtColumn;
    queue<pair<Node*, int>> q;
    q.push({root, 0});
    while (!q.empty()) {
        Node* node = q.front().first;
        int column = q.front().second;
        q.pop();
        valueAtColumn[column] = node->data; // later (lower) nodes replace earlier ones
        if (node->left) q.push({node->left, column - 1});
        if (node->right) q.push({node->right, column + 1});
    }
    vector<int> answer;
    for (auto& entry : valueAtColumn) answer.push_back(entry.second);
    return answer;
}

vector<int> topView(Node* root) {
